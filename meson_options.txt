option('build_date', type: 'string', value: '',
       description: 'Build date stamp shown in the About dialog (defaults to the configure-time date).')
option('http_backend', type: 'combo', choices: ['rust', 'curl'], value: 'rust',
       description: 'HTTP client used for page and subresource fetches. "rust" (default) is the in-tree Rust client in rust/http: HTTP/1.1 and HTTP/2 with its own framing and HPACK over OpenSSL TLS, pooled multiplexed connections, gzip/deflate/br/zstd decoding; proxied and FTP hops still go through libcurl. "curl" drives libcurl for every hop.')
option('quickjs', type: 'combo', choices: ['quickjs-ng', 'quickjs'], value: 'quickjs-ng',
       description: 'Which QuickJS the JavaScript binding runs on. "quickjs-ng" (default) builds the in-tree quickjs-ng fork in src/quickjs/, which carries the browser hooks. "quickjs" builds Fabrice Bellard\'s original QuickJS (https://github.com/bellard/quickjs), fetched at configure time through subprojects/quickjs.wrap and never vendored; src/ns_quickjs.h adapts the binding to its API and the few fork-only hooks fall back to portable approximations.')
option('gtk', type: 'feature', value: 'auto',
       description: 'Build the GTK 4 desktop shell (on by default when GTK 4 is present). Disable for a renderer-only build that needs no GTK 4 toolkit at all.')
option('wasm', type: 'feature', value: 'auto',
       description: 'Build the WebAssembly JS API over the vendored WAMR interpreter (on by default; disable on platforms WAMR does not support, e.g. NetBSD).')
option('avif', type: 'feature', value: 'auto',
       description: 'Decode AVIF images through libavif (on by default when libavif is present). AVIF is a rare format on the web and libavif pulls in a full AV1 decoder; disable to drop it.')
option('audio', type: 'feature', value: 'auto',
       description: 'Build the southstar-audio playback helper (SDL2 output + minimp3/pl_mpeg decode). Auto-detects SDL2; disable for engine-only builds without SDL2.')
option('webgpu', type: 'feature', value: 'auto',
       description: 'Build the experimental WebGPU API (navigator.gpu) over the external wgpu-native library. Auto: built whenever wgpu-native is found (pkg-config wgpu_native, or -Dwgpu_native_root pointing at an extracted release), skipped otherwise; -Dwebgpu=enabled hard-requires it, -Dwebgpu=disabled never builds it. Even when built, WebGPU stays off at runtime until the browser is started with --enable-webgpu (or NS_WEBGPU_ALLOW=1).')
option('wgpu_native_root', type: 'string', value: '',
       description: 'Root of an extracted wgpu-native release (expects include/webgpu/webgpu.h and lib/libwgpu_native.{so,a}). Used only when -Dwebgpu is enabled and pkg-config cannot find wgpu_native.')
option('ns-pango', type: 'feature', value: 'auto',
       description: 'Shape text through the bundled ns-pango fork, which caches finished glyph strings across layouts. Disabled links the system Pango instead, with src/ns_pango_names.h mapping the renamed API back.')
